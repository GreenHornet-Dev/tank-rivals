function drawGorillaChar(ctx, x, y, isP2, color) { ctx.save(); ctx.translate(x, y); const gc = color || (isP2 ? '#4488ff' : '#cc6600'); const gDark = shadeColor(gc, -40); const gDarker = shadeColor(gc, -60); const gLight = shadeColor(gc, 20); // Massive legs ctx.beginPath(); ctx.roundRect(-14, 8, 11, 18, 3); ctx.fillStyle = gDark; ctx.fill(); ctx.beginPath(); ctx.roundRect(3, 8, 11, 18, 3); ctx.fillStyle = gDark; ctx.fill(); // Feet ctx.beginPath(); ctx.ellipse(-9, 26, 8, 4, 0, 0, Math.PI*2); ctx.fillStyle = gDarker; ctx.fill(); ctx.beginPath(); ctx.ellipse(9, 26, 8, 4, 0, 0, Math.PI*2); ctx.fillStyle = gDarker; ctx.fill(); // Huge barrel chest / torso ctx.beginPath(); ctx.roundRect(-18, -10, 36, 22, 4); ctx.fillStyle = gc; ctx.fill(); // Chest highlight ctx.beginPath(); ctx.ellipse(0, -2, 10, 8, 0, 0, Math.PI*2); ctx.fillStyle = gLight; ctx.fill(); // Massive left arm (raised/muscular) ctx.beginPath(); ctx.moveTo(-18, -8); ctx.bezierCurveTo(-38, -12, -42, -28, -30, -32); ctx.bezierCurveTo(-26, -34, -22, -20, -18, -10); ctx.fillStyle = gDark; ctx.fill(); // Left fist ctx.beginPath(); ctx.arc(-31, -33, 7, 0, Math.PI*2); ctx.fillStyle = gDarker; ctx.fill(); // Massive right arm ctx.beginPath(); ctx.moveTo(18, -8); ctx.bezierCurveTo(38, -12, 42, -28, 30, -32); ctx.bezierCurveTo(26, -34, 22, -20, 18, -10); ctx.fillStyle = gDark; ctx.fill(); // Right fist ctx.beginPath(); ctx.arc(31, -33, 7, 0, Math.PI*2); ctx.fillStyle = gDarker; ctx.fill(); // Neck ctx.beginPath(); ctx.roundRect(-8, -18, 16, 10, 2); ctx.fillStyle = gc; ctx.fill(); // Big ape head ctx.beginPath(); ctx.ellipse(0, -30, 18, 16, 0, 0, Math.PI*2); ctx.fillStyle = gc; ctx.fill(); // Heavy brow ridge ctx.beginPath(); ctx.ellipse(0, -38, 16, 5, 0, 0, Math.PI*2); ctx.fillStyle = gDarker; ctx.fill(); // Brow shadow overhang ctx.beginPath(); ctx.rect(-16, -42, 32, 7); ctx.fillStyle = gDarker; ctx.fill(); // Deep-set eyes under brow ctx.beginPath(); ctx.arc(-6, -34, 3.5, 0, Math.PI*2); ctx.fillStyle = '#111'; ctx.fill(); ctx.beginPath(); ctx.arc(6, -34, 3.5, 0, Math.PI*2); ctx.fillStyle = '#111'; ctx.fill(); // Eye glint ctx.beginPath(); ctx.arc(-5, -35, 1.2, 0, Math.PI*2); ctx.fillStyle = 'white'; ctx.fill(); ctx.beginPath(); ctx.arc(7, -35, 1.2, 0, Math.PI*2); ctx.fillStyle = 'white'; ctx.fill(); // Wide flat nose ctx.beginPath(); ctx.ellipse(0, -27, 6, 4, 0, 0, Math.PI*2); ctx.fillStyle = gDarker; ctx.fill(); ctx.beginPath(); ctx.arc(-3, -26, 2, 0, Math.PI*2); ctx.fillStyle = '#222'; ctx.fill(); ctx.beginPath(); ctx.arc(3, -26, 2, 0, Math.PI*2); ctx.fillStyle = '#222'; ctx.fill(); // Wide jaw / muzzle ctx.beginPath(); ctx.ellipse(0, -20, 12, 7, 0, 0, Math.PI*2); ctx.fillStyle = gLight; ctx.fill(); // Snarling mouth ctx.beginPath(); ctx.moveTo(-8, -19); ctx.quadraticCurveTo(0, -14, 8, -19); ctx.strokeStyle = '#333'; ctx.lineWidth = 2; ctx.stroke(); // Fangs ctx.beginPath(); ctx.moveTo(-5, -19); ctx.lineTo(-4, -15); ctx.moveTo(5, -19); ctx.lineTo(4, -15); ctx.strokeStyle = 'white'; ctx.lineWidth = 1.5; ctx.stroke(); // Ear bumps ctx.beginPath(); ctx.arc(-17, -30, 5, 0, Math.PI*2); ctx.fillStyle = gc; ctx.fill(); ctx.beginPath(); ctx.arc(17, -30, 5, 0, Math.PI*2); ctx.fillStyle = gc; ctx.fill(); // Player color badge ctx.beginPath(); ctx.arc(0, -2, 4, 0, Math.PI*2); ctx.fillStyle = color; ctx.fill(); ctx.restore(); }
